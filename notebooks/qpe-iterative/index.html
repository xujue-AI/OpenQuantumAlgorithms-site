Iterative QPE: actual feedback instrument

This notebook constructs and evaluates measurement, reset and conditional phases. The static QDK estimator does not support this dynamic circuit. Use the locked environment or the public code package.

from pathlib import Path
import sys, os, math
root = Path.cwd()
if not (root / "src").is_dir() and (root.parent / "src").is_dir():
    root = root.parent
if (root / "src").is_dir():
    sys.path.insert(0, str(root / "src"))
try:
    from openquantum.qft_qpe import *
except ModuleNotFoundError:
    from qft_qpe import *  # Download qft_qpe.py beside this notebook.
from qiskit.quantum_info import Operator
import matplotlib.pyplot as plt
from IPython.display import display
from matplotlib_inline.backend_inline import set_matplotlib_formats
set_matplotlib_formats("svg")

phase, m = 0.3, 5
u, prep = phase_gate_example(phase)
instrument = build_iterative_qpe(m, u, prep)
display(instrument.draw(output="mpl", fold=36)); plt.close("all")
